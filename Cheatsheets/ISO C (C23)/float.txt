//|----------------------------------------------------------------------------
//| Radix and Rounding (float.h)
//|----------------------------------------------------------------------------

FLT_RADIX                                                                          /* Base (radix) used to represent the exponent in floating-point types. */
FLT_ROUNDS                                                                         /* Current rounding mode for floating-point addition. */
FLT_EVAL_METHOD                                                                    /* Indicates how floating-point expressions are evaluated. */


//|----------------------------------------------------------------------------
//| Precision (float.h)
//|----------------------------------------------------------------------------

FLT_MANT_DIG                                                                       /* Number of base-FLT_RADIX digits in the float mantissa. */
DBL_MANT_DIG                                                                       /* Number of base-FLT_RADIX digits in the double mantissa. */
LDBL_MANT_DIG                                                                      /* Number of base-FLT_RADIX digits in the long double mantissa. */
FLT_DIG                                                                            /* Number of decimal digits of precision for float. */
DBL_DIG                                                                            /* Number of decimal digits of precision for double. */
LDBL_DIG                                                                           /* Number of decimal digits of precision for long double. */


//|----------------------------------------------------------------------------
//| Exponent Range (float.h)
//|----------------------------------------------------------------------------

FLT_MIN_EXP                                                                        /* Minimum negative integer exponent for float. */
DBL_MIN_EXP                                                                        /* Minimum negative integer exponent for double. */
LDBL_MIN_EXP                                                                       /* Minimum negative integer exponent for long double. */
FLT_MIN_10_EXP                                                                     /* Minimum negative decimal exponent for float. */
DBL_MIN_10_EXP                                                                     /* Minimum negative decimal exponent for double. */
LDBL_MIN_10_EXP                                                                    /* Minimum negative decimal exponent for long double. */
FLT_MAX_EXP                                                                        /* Maximum integer exponent for float. */
DBL_MAX_EXP                                                                        /* Maximum integer exponent for double. */
LDBL_MAX_EXP                                                                       /* Maximum integer exponent for long double. */
FLT_MAX_10_EXP                                                                     /* Maximum decimal exponent for float. */
DBL_MAX_10_EXP                                                                     /* Maximum decimal exponent for double. */
LDBL_MAX_10_EXP                                                                    /* Maximum decimal exponent for long double. */


//|----------------------------------------------------------------------------
//| Range and Precision Limits (float.h)
//|----------------------------------------------------------------------------

FLT_MAX                                                                            /* Maximum finite representable value for float. */
DBL_MAX                                                                            /* Maximum finite representable value for double. */
LDBL_MAX                                                                           /* Maximum finite representable value for long double. */
FLT_EPSILON                                                                        /* Smallest positive float x such that 1.0 + x != 1.0. */
DBL_EPSILON                                                                        /* Smallest positive double x such that 1.0 + x != 1.0. */
LDBL_EPSILON                                                                       /* Smallest positive long double x such that 1.0 + x != 1.0. */
FLT_MIN                                                                            /* Minimum positive normalized value for float. */
DBL_MIN                                                                            /* Minimum positive normalized value for double. */
LDBL_MIN                                                                           /* Minimum positive normalized value for long double. */


//|----------------------------------------------------------------------------
//| Subnormal Support (float.h) — C11+
//|----------------------------------------------------------------------------

FLT_HAS_SUBNORM                                                                    /* Indicates whether float supports subnormal (denormalized) numbers. */
DBL_HAS_SUBNORM                                                                    /* Indicates whether double supports subnormal numbers. */
LDBL_HAS_SUBNORM                                                                   /* Indicates whether long double supports subnormal numbers. */
FLT_TRUE_MIN                                                                       /* Minimum positive value for float, including subnormals. */
DBL_TRUE_MIN                                                                       /* Minimum positive value for double, including subnormals. */
LDBL_TRUE_MIN                                                                      /* Minimum positive value for long double, including subnormals. */


//|----------------------------------------------------------------------------
//| Decimal Digit Requirements (float.h) — C11+
//|----------------------------------------------------------------------------

FLT_DECIMAL_DIG                                                                    /* Number of decimal digits needed to uniquely represent any float value. */
DBL_DECIMAL_DIG                                                                    /* Number of decimal digits needed to uniquely represent any double value. */
LDBL_DECIMAL_DIG                                                                   /* Number of decimal digits needed to uniquely represent any long double value. */
DECIMAL_DIG                                                                        /* Number of decimal digits needed to uniquely represent any value of the widest supported floating-point type. */

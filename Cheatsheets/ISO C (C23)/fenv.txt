//| Floating-Point Environment (fenv.h) — C99+

int           feholdexcept     (fenv_t *envp);                                    /* Saves the environment and clears exception flags. */
int           feupdateenv      (const fenv_t *envp);                              /* Restores the environment while preserving currently raised exceptions. */

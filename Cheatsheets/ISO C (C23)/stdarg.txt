//|----------------------------------------------------------------------------
//| Variable Argument Macros (stdarg.h)
//|----------------------------------------------------------------------------

void          va_start      (va_list ap, last_param);                            /* Macro. Initializes ap to retrieve arguments after last_param (the last named parameter). */
type          va_arg        (va_list ap, type);                                  /* Macro. Retrieves the next argument, interpreted as type, from ap and advances ap. */
void          va_end        (va_list ap);                                        /* Macro. Performs cleanup for an ap initialized by va_start or va_copy. */
void          va_copy       (va_list dest, va_list src);                         /* Macro (C99+). Copies the current state of src into dest. */


//|----------------------------------------------------------------------------
//| Types (stdarg.h)
//|----------------------------------------------------------------------------

va_list                                                                            /* Type suitable for holding information needed to retrieve variable arguments. */


//|----------------------------------------------------------------------------
//| Notes (stdarg.h)
//|----------------------------------------------------------------------------
/* va_start, va_arg, va_end, and va_copy are macros, not true functions;
   their "signatures" above are illustrative. va_arg's second argument is a
   type name (not a value), and va_start's second argument must be the name
   of the last named parameter before the variable argument list — these
   cannot be expressed as an ordinary C function prototype. */

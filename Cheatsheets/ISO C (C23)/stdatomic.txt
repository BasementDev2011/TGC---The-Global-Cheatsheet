//|----------------------------------------------------------------------------
//| Generic Atomic Operations (stdatomic.h) — C11+, optional
//|----------------------------------------------------------------------------
/* A = an atomic type (e.g. atomic_int); C = the corresponding non-atomic
   value type (e.g. int); M = the operand type used for arithmetic. These
   are type-generic operations (implemented as macros/generic functions in
   the standard) that work across the whole family of atomic types. */

void          atomic_init                          (volatile A *obj, C value);                    /* Initializes an atomic object obj (non-atomically) to value. */
_Bool         atomic_is_lock_free                  (const volatile A *obj);                        /* Checks whether operations on obj are lock-free. */

void          atomic_store                         (volatile A *obj, C value);                     /* Atomically stores value into obj (sequentially consistent). */
void          atomic_store_explicit                (volatile A *obj, C value, memory_order order); /* Atomically stores value into obj using the specified memory order. */

C             atomic_load                          (volatile A *obj);                              /* Atomically loads and returns the value of obj (sequentially consistent). */
C             atomic_load_explicit                 (volatile A *obj, memory_order order);          /* Atomically loads obj using the specified memory order. */

C             atomic_exchange                      (volatile A *obj, C value);                     /* Atomically replaces obj with value, returning the previous value. */
C             atomic_exchange_explicit             (volatile A *obj, C value, memory_order order); /* Atomically exchanges obj with the specified memory order. */

_Bool         atomic_compare_exchange_strong       (volatile A *obj, C *expected, C desired);      /* Atomically compares obj to *expected; if equal, sets obj to desired. */
_Bool         atomic_compare_exchange_strong_explicit (volatile A *obj, C *expected, C desired, memory_order success, memory_order failure); /* Strong compare-exchange with explicit memory orders. */
_Bool         atomic_compare_exchange_weak         (volatile A *obj, C *expected, C desired);      /* Like the strong version, but may fail spuriously. */
_Bool         atomic_compare_exchange_weak_explicit   (volatile A *obj, C *expected, C desired, memory_order success, memory_order failure); /* Weak compare-exchange with explicit memory orders. */

C             atomic_fetch_add                     (volatile A *obj, M operand);                   /* Atomically adds operand to obj, returning the previous value. */
C             atomic_fetch_add_explicit            (volatile A *obj, M operand, memory_order order); /* Atomic add with an explicit memory order. */
C             atomic_fetch_sub                     (volatile A *obj, M operand);                   /* Atomically subtracts operand from obj, returning the previous value. */
C             atomic_fetch_sub_explicit            (volatile A *obj, M operand, memory_order order); /* Atomic subtract with an explicit memory order. */
C             atomic_fetch_or                      (volatile A *obj, M operand);                   /* Atomically ORs operand into obj, returning the previous value. */
C             atomic_fetch_or_explicit             (volatile A *obj, M operand, memory_order order); /* Atomic OR with an explicit memory order. */
C             atomic_fetch_xor                     (volatile A *obj, M operand);                   /* Atomically XORs operand into obj, returning the previous value. */
C             atomic_fetch_xor_explicit            (volatile A *obj, M operand, memory_order order); /* Atomic XOR with an explicit memory order. */
C             atomic_fetch_and                     (volatile A *obj, M operand);                   /* Atomically ANDs operand into obj, returning the previous value. */
C             atomic_fetch_and_explicit            (volatile A *obj, M operand, memory_order order); /* Atomic AND with an explicit memory order. */


//|----------------------------------------------------------------------------

_Bool         atomic_flag_test_and_set             (volatile atomic_flag *obj);                    /* Atomically sets obj to true, returning its previous value. */
_Bool         atomic_flag_test_and_set_explicit    (volatile atomic_flag *obj, memory_order order); /* Atomic test-and-set with an explicit memory order. */
void          atomic_flag_clear                    (volatile atomic_flag *obj);                    /* Atomically sets obj to false. */
void          atomic_flag_clear_explicit           (volatile atomic_flag *obj, memory_order order); /* Atomic clear with an explicit memory order. */

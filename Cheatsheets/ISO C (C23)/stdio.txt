//|----------------------------------------------------------------------------
//| File Access (stdio.h)
//|----------------------------------------------------------------------------

FILE         *fopen         (const char *restrict filename, const char *restrict mode);   /* Opens a file and associates it with a stream. */
FILE         *freopen       (const char *restrict filename, const char *restrict mode, FILE *restrict stream); /* Reopens a stream with a different file or mode. */
int           fclose        (FILE *stream);                                       /* Closes a stream. */
int           fflush        (FILE *stream);                                       /* Flushes a stream's output buffer. */
void          setbuf        (FILE *restrict stream, char *restrict buffer);       /* Sets the buffer for a stream. */
int           setvbuf       (FILE *restrict stream, char *restrict buffer, int mode, size_t size); /* Sets the buffer and buffering mode for a stream. */


//|----------------------------------------------------------------------------
//| Formatted Output (stdio.h)
//|----------------------------------------------------------------------------

int           printf        (const char *restrict format, ...);                  /* Writes formatted output to stdout. */
int           fprintf       (FILE *restrict stream, const char *restrict format, ...); /* Writes formatted output to a stream. */
int           sprintf       (char *restrict buffer, const char *restrict format, ...); /* Writes formatted output to a string. */
int           snprintf      (char *restrict buffer, size_t size, const char *restrict format, ...); /* Writes formatted output to a string with a size limit. */
int           vprintf       (const char *restrict format, va_list args);          /* Writes formatted output to stdout using a variable argument list. */
int           vfprintf      (FILE *restrict stream, const char *restrict format, va_list args); /* Writes formatted output to a stream using a variable argument list. */
int           vsprintf      (char *restrict buffer, const char *restrict format, va_list args); /* Writes formatted output to a string using a variable argument list. */
int           vsnprintf     (char *restrict buffer, size_t size, const char *restrict format, va_list args); /* Writes formatted output to a string with a size limit using a variable argument list. */


//|----------------------------------------------------------------------------
//| Formatted Input (stdio.h)
//|----------------------------------------------------------------------------

int           scanf         (const char *restrict format, ...);                  /* Reads formatted input from stdin. */
int           fscanf        (FILE *restrict stream, const char *restrict format, ...); /* Reads formatted input from a stream. */
int           sscanf        (const char *restrict buffer, const char *restrict format, ...); /* Reads formatted input from a string. */
int           vscanf        (const char *restrict format, va_list args);          /* Reads formatted input from stdin using a variable argument list. */
int           vfscanf       (FILE *restrict stream, const char *restrict format, va_list args); /* Reads formatted input from a stream using a variable argument list. */
int           vsscanf       (const char *restrict buffer, const char *restrict format, va_list args); /* Reads formatted input from a string using a variable argument list. */


//|----------------------------------------------------------------------------
//| Character Input/Output (stdio.h)
//|----------------------------------------------------------------------------

int           fgetc         (FILE *stream);                                       /* Reads a character from a stream. */
char         *fgets         (char *restrict str, int count, FILE *restrict stream); /* Reads a line from a stream. */
int           fputc         (int ch, FILE *stream);                               /* Writes a character to a stream. */
int           fputs         (const char *restrict str, FILE *restrict stream);    /* Writes a string to a stream. */
int           getc          (FILE *stream);                                       /* Reads a character from a stream (may be a macro). */
int           getchar       (void);                                               /* Reads a character from stdin. */
char         *gets          (char *str);                                          /* LEGACY — removed in C11, unsafe (no bounds checking). Reads a line from stdin into str. */
int           putc          (int ch, FILE *stream);                               /* Writes a character to a stream (may be a macro). */
int           putchar       (int ch);                                             /* Writes a character to stdout. */
int           puts          (const char *str);                                    /* Writes a string to stdout followed by a newline. */
int           ungetc        (int ch, FILE *stream);                               /* Pushes a character back onto a stream. */


//|----------------------------------------------------------------------------
//| Direct Input/Output (stdio.h)
//|----------------------------------------------------------------------------

size_t        fread         (void *restrict buffer, size_t size, size_t count, FILE *restrict stream); /* Reads binary data from a stream. */
size_t        fwrite        (const void *restrict buffer, size_t size, size_t count, FILE *restrict stream); /* Writes binary data to a stream. */


//|----------------------------------------------------------------------------
//| File Positioning (stdio.h)
//|----------------------------------------------------------------------------

int           fgetpos       (FILE *restrict stream, fpos_t *restrict pos);        /* Gets the current file position of a stream. */
int           fsetpos       (FILE *stream, const fpos_t *pos);                    /* Sets the current file position of a stream. */
int           fseek         (FILE *stream, long int offset, int origin);          /* Sets the file position of a stream. */
long int      ftell         (FILE *stream);                                       /* Returns the current file position of a stream. */
void          rewind        (FILE *stream);                                       /* Resets the file position to the beginning of a stream. */


//|----------------------------------------------------------------------------
//| Error Handling (stdio.h)
//|----------------------------------------------------------------------------

void          clearerr      (FILE *stream);                                       /* Clears the end-of-file and error indicators for a stream. */
int           feof          (FILE *stream);                                       /* Checks the end-of-file indicator for a stream. */
int           ferror        (FILE *stream);                                       /* Checks the error indicator for a stream. */
void          perror        (const char *str);                                    /* Prints a description of the current error to stderr. */


//|----------------------------------------------------------------------------
//| File Operations (stdio.h)
//|----------------------------------------------------------------------------

int           remove        (const char *filename);                               /* Deletes a file. */
int           rename        (const char *old_filename, const char *new_filename);  /* Renames a file. */
FILE         *tmpfile       (void);                                                /* Creates and opens a temporary file. */
char         *tmpnam        (char *str);                                           /* LEGACY — unsafe, avoid; use tmpfile instead. Generates a unique temporary filename. */


//|----------------------------------------------------------------------------
//| Types (stdio.h)
//|----------------------------------------------------------------------------

FILE                                                                              /* Type representing an open file stream. */
fpos_t                                                                            /* Type representing a file position. */
size_t                                                                            /* Unsigned integer type used for sizes. */


//|----------------------------------------------------------------------------
//| Constants and Macros (stdio.h)
//|----------------------------------------------------------------------------

NULL                                                                              /* Null pointer constant. */
EOF                                                                               /* Integer constant indicating end-of-file. */
BUFSIZ                                                                            /* Default buffer size used by setbuf. */
FOPEN_MAX                                                                         /* Minimum number of files that can be open simultaneously. */
FILENAME_MAX                                                                      /* Maximum length of a filename string. */
L_tmpnam                                                                          /* LEGACY. Size needed for an array to hold a tmpnam result. */
TMP_MAX                                                                           /* LEGACY. Minimum number of unique filenames tmpnam can generate. */
_IOFBF                                                                            /* Full buffering mode constant for setvbuf. */
_IOLBF                                                                            /* Line buffering mode constant for setvbuf. */
_IONBF                                                                            /* No buffering mode constant for setvbuf. */
SEEK_SET                                                                          /* Seek from the beginning of the file. */
SEEK_CUR                                                                          /* Seek from the current file position. */
SEEK_END                                                                          /* Seek from the end of the file. */
stdin                                                                             /* Standard input stream. */
stdout                                                                            /* Standard output stream. */
stderr                                                                            /* Standard error stream. */

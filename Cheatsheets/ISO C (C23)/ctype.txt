//|----------------------------------------------------------------------------
//| Character Classification (ctype.h)
//|----------------------------------------------------------------------------

int           isalnum       (int ch);                                            /* Checks if a character is alphanumeric. */
int           isalpha       (int ch);                                            /* Checks if a character is alphabetic. */
int           isblank       (int ch);                                            /* C99+. Checks if a character is blank (space or tab). */
int           iscntrl       (int ch);                                            /* Checks if a character is a control character. */
int           isdigit       (int ch);                                            /* Checks if a character is a decimal digit. */
int           isgraph       (int ch);                                            /* Checks if a character has a graphical representation. */
int           islower       (int ch);                                            /* Checks if a character is lowercase. */
int           isprint       (int ch);                                            /* Checks if a character is printable. */
int           ispunct       (int ch);                                            /* Checks if a character is punctuation. */
int           isspace       (int ch);                                            /* Checks if a character is whitespace. */
int           isupper       (int ch);                                            /* Checks if a character is uppercase. */
int           isxdigit      (int ch);                                            /* Checks if a character is a hexadecimal digit. */


//|----------------------------------------------------------------------------
//| Character Conversion (ctype.h)
//|----------------------------------------------------------------------------

int           tolower       (int ch);                                            /* Converts a character to lowercase. */
int           toupper       (int ch);                                            /* Converts a character to uppercase. */


//|----------------------------------------------------------------------------
//| Notes (ctype.h)
//|----------------------------------------------------------------------------
/* All functions in this header require the argument to be representable as
   unsigned char or equal to EOF; passing other values is undefined behavior.
   Classification and conversion results are affected by the current locale
   (see locale.h / setlocale). */

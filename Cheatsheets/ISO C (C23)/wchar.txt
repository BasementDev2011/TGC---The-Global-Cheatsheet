int           wmemcmp       (const wchar_t *s1, const wchar_t *s2, size_t n);    /* Compares the first n wide characters of two arrays. */

/* wcsnlen_s, wcrtomb_s, mbsrtowcs_s, wcsrtombs_s, and formatted-I/O _s       */
/* variants — not guaranteed present.                                        */
/*                                                                            */

/* wcsncasecmp, etc.) are not part of ISO C <wchar.h> and are excluded.       */

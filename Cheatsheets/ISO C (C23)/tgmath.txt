//|----------------------------------------------------------------------------
//| Notes (tgmath.h)
//|----------------------------------------------------------------------------

/* real-number-expression: a placeholder for an expression of real floating   */
/* type (float, double, or long double, after usual arithmetic conversions).  */
/*                                                                            */
/* complex-or-real-expression: a placeholder for an expression of either a    */
/* real floating type or a complex floating type (float _Complex,            */
/* double _Complex, or long double _Complex).                                */
/*                                                                            */
/* <tgmath.h> includes <math.h> and, if supported, <complex.h>. Each entry    */
/* above is a function-like macro (not a real function) that inspects the    */
/* type(s) of its argument(s) at compile time and expands to a call to the    */
/* matching real or complex, float/double/long double suffixed function.      */
/* If __STDC_NO_COMPLEX__ is defined, complex support and <complex.h> are     */
/* absent, and the complex-dispatch macros resolve only to their real forms.  */
/* This header defines no functions, types, structs, or constants of its own */
/* beyond these type-generic macros.                                         */

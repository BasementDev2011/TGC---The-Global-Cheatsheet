//|----------------------------------------------------------------------------
//| Type-Generic Classification and Comparison Macros (math.h) — C99+
//|----------------------------------------------------------------------------
/* These are function-like macros defined by the standard to accept any
   real floating type (float, double, or long double) — and in some cases
   any real type — without committing to one specific parameter type. In
   this cheatsheet, "real-number-expression" is used as a placeholder to
   mean: an expression of any real floating (or real) type is accepted
   here; the macro selects the correct behavior based on the argument's
   type at compile time. */

int           fpclassify       (real-number-expression x);                       /* Classifies x as NaN, infinite, normal, subnormal, or zero. */
int           isfinite         (real-number-expression x);                       /* Checks whether x is finite. */
int           isinf            (real-number-expression x);                       /* Checks whether x is infinite. */
int           isnan            (real-number-expression x);                       /* Checks whether x is NaN. */
int           isnormal         (real-number-expression x);                       /* Checks whether x is a normal (not zero, subnormal, infinite, or NaN) value. */
int           signbit          (real-number-expression x);                       /* Checks whether the sign of x is negative. */
int           isgreater        (real-number-expression x, real-number-expression y);      /* Checks whether x > y, without raising an exception for unordered (NaN) operands. */
int           isgreaterequal   (real-number-expression x, real-number-expression y);      /* Checks whether x >= y, without raising an exception for unordered operands. */
int           isless           (real-number-expression x, real-number-expression y);      /* Checks whether x < y, without raising an exception for unordered operands. */
int           islessequal      (real-number-expression x, real-number-expression y);      /* Checks whether x <= y, without raising an exception for unordered operands. */
int           islessgreater    (real-number-expression x, real-number-expression y);      /* Checks whether x < y or x > y, without raising an exception for unordered operands. */
int           isunordered      (real-number-expression x, real-number-expression y);      /* Checks whether x and y are unordered (i.e. at least one is NaN). */

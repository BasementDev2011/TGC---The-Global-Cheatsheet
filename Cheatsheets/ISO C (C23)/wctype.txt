//|----------------------------------------------------------------------------
//| Wide Character Types (wctype.h)
//|----------------------------------------------------------------------------

wint_t                                                                         /* Integer type capable of representing any wide character value or WEOF. */
wctype_t                                                                       /* Type identifying a wide-character classification. */
wctrans_t                                                                      /* Type identifying a wide-character mapping. */

//|----------------------------------------------------------------------------
//| Wide Character Classification (wctype.h)
//|----------------------------------------------------------------------------

int     iswalnum  (wint_t wc);                                                  /* Tests whether a wide character is alphanumeric. */
int     iswalpha  (wint_t wc);                                                  /* Tests whether a wide character is alphabetic. */
int     iswblank  (wint_t wc);                                                  /* (C99) Tests whether a wide character is blank. */
int     iswcntrl  (wint_t wc);                                                  /* Tests whether a wide character is a control character. */
int     iswdigit  (wint_t wc);                                                  /* Tests whether a wide character is a decimal digit. */
int     iswgraph  (wint_t wc);                                                  /* Tests whether a wide character is graphical. */
int     iswlower  (wint_t wc);                                                  /* Tests whether a wide character is lowercase. */
int     iswprint  (wint_t wc);                                                  /* Tests whether a wide character is printable. */
int     iswpunct  (wint_t wc);                                                  /* Tests whether a wide character is punctuation. */
int     iswspace  (wint_t wc);                                                  /* Tests whether a wide character is whitespace. */
int     iswupper  (wint_t wc);                                                  /* Tests whether a wide character is uppercase. */
int     iswxdigit (wint_t wc);                                                  /* Tests whether a wide character is a hexadecimal digit. */

//|----------------------------------------------------------------------------
//| Extensible Wide Character Classification (wctype.h)
//|----------------------------------------------------------------------------

wctype_t wctype   (const char *property);                                       /* Returns a descriptor for a named wide-character classification. */
int      iswctype (wint_t wc, wctype_t desc);                                   /* Tests whether a wide character has the classification described by desc. */

//|----------------------------------------------------------------------------
//| Wide Character Case Mapping (wctype.h)
//|----------------------------------------------------------------------------

wint_t towlower (wint_t wc);                                                     /* Converts an uppercase wide character to lowercase. */
wint_t towupper (wint_t wc);                                                     /* Converts a lowercase wide character to uppercase. */

//|----------------------------------------------------------------------------
//| Extensible Wide Character Case Mapping (wctype.h)
//|----------------------------------------------------------------------------

wctrans_t wctrans   (const char *property);                                     /* Returns a descriptor for a named wide-character mapping. */
wint_t    towctrans (wint_t wc, wctrans_t desc);                                /* Applies the mapping described by desc to a wide character. */

//|----------------------------------------------------------------------------
//| Macros (wctype.h)
//|----------------------------------------------------------------------------

WEOF                                                                            /* Wide-character end-of-file value that is not a valid wide character. */

//|----------------------------------------------------------------------------
//| Thread Management (threads.h)
//|----------------------------------------------------------------------------

int             thrd_create   (thrd_t *thr, thrd_start_t func, void *arg);          /* Creates a new thread running func(arg). */
thrd_t          thrd_current  (void);                                               /* Returns the identifier of the calling thread. */
int             thrd_detach   (thrd_t thr);                                         /* Detaches a thread so its resources are released on termination. */
int             thrd_equal    (thrd_t thr0, thrd_t thr1);                           /* Compares two thread identifiers for equality. */
_Noreturn void  thrd_exit     (int res);                                            /* Terminates the calling thread, returning res. */
int             thrd_join     (thrd_t thr, int *res);                               /* Blocks until thr terminates, optionally storing its result in res. */
int             thrd_sleep    (const struct timespec *duration, struct timespec *remaining); /* Suspends the calling thread for a given duration. */
void            thrd_yield    (void);                                               /* Yields execution to another thread. */

//|----------------------------------------------------------------------------
//| Thread Return Codes (threads.h)
//|----------------------------------------------------------------------------

thrd_success                                                                        /* Indicates a function succeeded. */
thrd_busy                                                                           /* Indicates a resource is temporarily unavailable. */
thrd_error                                                                          /* Indicates an error occurred. */
thrd_nomem                                                                          /* Indicates memory allocation failed. */
thrd_timedout                                                                       /* Indicates a timed operation expired before success. */

//|----------------------------------------------------------------------------
//| Mutexes (threads.h)
//|----------------------------------------------------------------------------

int           mtx_init      (mtx_t *mtx, int type);                                 /* Initializes a mutex of the given type. */
int           mtx_lock      (mtx_t *mtx);                                           /* Blocks until the calling thread locks mtx. */
int           mtx_timedlock (mtx_t *restrict mtx, const struct timespec *restrict ts); /* Blocks until mtx is locked or ts is reached. */
int           mtx_trylock   (mtx_t *mtx);                                           /* Attempts to lock mtx without blocking. */
int           mtx_unlock    (mtx_t *mtx);                                           /* Unlocks a previously locked mtx. */
void          mtx_destroy   (mtx_t *mtx);                                           /* Releases resources used by mtx. */

//|----------------------------------------------------------------------------
//| Mutex Type Constants (threads.h)
//|----------------------------------------------------------------------------

mtx_plain                                                                           /* A simple, non-recursive, non-timed mutex. */
mtx_recursive                                                                       /* A recursive mutex. */
mtx_timed                                                                           /* A mutex that supports timeout on locking. */

//|----------------------------------------------------------------------------
//| Condition Variables (threads.h)
//|----------------------------------------------------------------------------

int           cnd_broadcast (cnd_t *cond);                                          /* Wakes all threads blocked on cond. */
void          cnd_destroy   (cnd_t *cond);                                          /* Releases resources used by cond. */
int           cnd_init      (cnd_t *cond);                                          /* Initializes a condition variable. */
int           cnd_signal    (cnd_t *cond);                                          /* Wakes one thread blocked on cond. */
int           cnd_timedwait (cnd_t *restrict cond, mtx_t *restrict mtx, const struct timespec *restrict ts); /* Blocks on cond until signaled or ts is reached. */
int           cnd_wait      (cnd_t *cond, mtx_t *mtx);                              /* Blocks the calling thread on cond, releasing mtx. */

void          tss_delete    (tss_t key);                                            /* Deletes a thread-specific storage key. */
void         *tss_get       (tss_t key);                                            /* Returns the value associated with key for the calling thread. */
int           tss_set       (tss_t key, void *val);                                 /* Sets the value associated with key for the calling thread. */

//|----------------------------------------------------------------------------
//| Notes (threads.h)
//|----------------------------------------------------------------------------

/* Threading is a CONDITIONAL FEATURE in C11 (Annex E). A conforming          */
/* implementation need not provide <threads.h> at all; if unsupported, the    */
/* macro __STDC_NO_THREADS__ is defined.                                      */
/*                                                                            */
/* struct timespec (used by thrd_sleep, mtx_timedlock, cnd_timedwait) is      */
/* defined in <time.h>, not by this header.                                   */
/*                                                                            */
/* POSIX pthread_* functions (<pthread.h>) are a separate, non-ISO C API and  */
/* are not part of <threads.h>; they are excluded from this cheatsheet.       */

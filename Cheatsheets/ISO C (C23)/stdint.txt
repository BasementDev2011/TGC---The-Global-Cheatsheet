//|----------------------------------------------------------------------------

/* Only defined if the implementation has an integer type with that exact
   width and two's-complement (or matching) representation. */

int8_t                                                                             /* Signed integer type of exactly 8 bits. */
int16_t                                                                            /* Signed integer type of exactly 16 bits. */
int32_t                                                                            /* Signed integer type of exactly 32 bits. */
int64_t                                                                            /* Signed integer type of exactly 64 bits. */
uint8_t                                                                            /* Unsigned integer type of exactly 8 bits. */
uint16_t                                                                           /* Unsigned integer type of exactly 16 bits. */
uint32_t                                                                           /* Unsigned integer type of exactly 32 bits. */
uint64_t                                                                           /* Unsigned integer type of exactly 64 bits. */

int_least32_t                                                                      /* Smallest signed integer type with at least 32 bits. */
int_least64_t                                                                      /* Smallest signed integer type with at least 64 bits. */
uint_least8_t                                                                      /* Smallest unsigned integer type with at least 8 bits. */
uint_least16_t                                                                     /* Smallest unsigned integer type with at least 16 bits. */
uint_least32_t                                                                     /* Smallest unsigned integer type with at least 32 bits. */
uint_least64_t                                                                     /* Smallest unsigned integer type with at least 64 bits. */

//| Fastest Minimum-Width Integer Types (stdint.h) — C99+
//|----------------------------------------------------------------------------

int_fast16_t                                                                       /* Fastest signed integer type with at least 16 bits. */
int_fast32_t                                                                       /* Fastest signed integer type with at least 32 bits. */
int_fast64_t                                                                       /* Fastest signed integer type with at least 64 bits. */
uint_fast8_t                                                                       /* Fastest unsigned integer type with at least 8 bits. */
uint_fast16_t                                                                      /* Fastest unsigned integer type with at least 16 bits. */
uint_fast32_t                                                                      /* Fastest unsigned integer type with at least 32 bits. */
uint_fast64_t                                                                      /* Fastest unsigned integer type with at least 64 bits. */


//|----------------------------------------------------------------------------
//| Integer Types Capable of Holding Pointers (stdint.h) — C99+, OPTIONAL
//|----------------------------------------------------------------------------

intptr_t                                                                           /* Signed integer type large enough to hold a pointer. */
uintptr_t                                                                          /* Unsigned integer type large enough to hold a pointer. */


//|----------------------------------------------------------------------------
//| Greatest-Width Integer Types (stdint.h) — C99+
//|----------------------------------------------------------------------------

intmax_t                                                                           /* Signed integer type capable of representing any value of any signed integer type. */
uintmax_t                                                                          /* Unsigned integer type capable of representing any value of any unsigned integer type. */

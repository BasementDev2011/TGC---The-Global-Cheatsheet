clock_t       clock         (void);                                              /* Returns the processor time used by the program so far. */
double        difftime      (time_t time1, time_t time0);                        /* Computes the difference, in seconds, between two calendar times. */
time_t        mktime        (struct tm *timeptr);                                /* Converts a broken-down local time into a calendar time. */
time_t        time          (time_t *timer);                                     /* Returns the current calendar time, optionally storing it via timer. */
int           timespec_get  (struct timespec *ts, int base);                     /* (C11) Sets ts to the current calendar time, per the given time base. */

//|----------------------------------------------------------------------------
//| Time Conversion (time.h)
//|----------------------------------------------------------------------------

char         *asctime       (const struct tm *timeptr);                          /* Converts a broken-down time into a fixed-format string. */
char         *ctime         (const time_t *timer);                               /* Converts a calendar time into a fixed-format local time string. */
struct tm    *gmtime        (const time_t *timer);                               /* Converts a calendar time into broken-down UTC time. */
struct tm    *localtime     (const time_t *timer);                               /* Converts a calendar time into broken-down local time. */
size_t        strftime      (char *restrict s, size_t maxsize, const char *restrict format, const struct tm *restrict timeptr); /* Formats a broken-down time as a string. */

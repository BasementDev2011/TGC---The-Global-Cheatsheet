//|----------------------------------------------------------------------------
//| Variable Arguments (stdarg.h)
//|----------------------------------------------------------------------------

void          va_start      (va_list ap, parmN);                                 /* Initializes ap for use by va_arg and va_end; parmN is the last named parameter before the variable argument list. */
type          va_arg        (va_list ap, type);                                  /* Expands to the next argument in the list ap, with the given type, and advances ap. */
void          va_end        (va_list ap);                                        /* Performs cleanup for ap, required before the function returns. */


//|----------------------------------------------------------------------------
//| Types (stdarg.h)
//|----------------------------------------------------------------------------

va_list                                                                         /* Type suitable for holding information needed by va_start, va_arg, and va_end. */

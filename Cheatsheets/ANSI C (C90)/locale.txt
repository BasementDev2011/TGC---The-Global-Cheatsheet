//|----------------------------------------------------------------------------
//| Locale Control (locale.h)
//|----------------------------------------------------------------------------

char         *setlocale     (int category, const char *locale);                  /* Sets or queries the program's current locale for the given category. */
struct lconv *localeconv    (void);                                              /* Returns a pointer to a struct lconv populated with the current locale's numeric formatting conventions. */


//|----------------------------------------------------------------------------
//| Locale Category Constants (locale.h)
//|----------------------------------------------------------------------------

LC_ALL                                                                          /* Selects the entire locale. */
LC_COLLATE                                                                      /* Selects the locale category affecting strcoll and strxfrm. */
LC_CTYPE                                                                        /* Selects the locale category affecting character handling functions. */
LC_MONETARY                                                                     /* Selects the locale category affecting monetary formatting in localeconv. */
LC_NUMERIC                                                                      /* Selects the locale category affecting decimal-point and numeric formatting. */
LC_TIME                                                                         /* Selects the locale category affecting strftime. */


//|----------------------------------------------------------------------------
//| Constants (locale.h)
//|----------------------------------------------------------------------------

NULL                                                                            /* Null pointer constant. */


//|----------------------------------------------------------------------------
//| Types (locale.h)
//|----------------------------------------------------------------------------

struct lconv                                                                    /* Structure containing numeric and monetary formatting conventions for the current locale. */


//|----------------------------------------------------------------------------
//| struct lconv Members (locale.h)
//|----------------------------------------------------------------------------

decimal_point                                                                   /* Decimal-point character for non-monetary quantities. */
thousands_sep                                                                   /* Thousands separator for non-monetary quantities. */
grouping                                                                        /* Digit grouping pattern for non-monetary quantities. */
int_curr_symbol                                                                 /* International currency symbol. */
currency_symbol                                                                 /* Local currency symbol. */
mon_decimal_point                                                               /* Decimal-point character for monetary quantities. */
mon_thousands_sep                                                               /* Thousands separator for monetary quantities. */
mon_grouping                                                                    /* Digit grouping pattern for monetary quantities. */
positive_sign                                                                   /* String indicating a nonnegative monetary value. */
negative_sign                                                                   /* String indicating a negative monetary value. */
int_frac_digits                                                                 /* Number of fractional digits for internationally formatted monetary values. */
frac_digits                                                                     /* Number of fractional digits for locally formatted monetary values. */
p_cs_precedes                                                                   /* Whether currency_symbol precedes a nonnegative value. */
p_sep_by_space                                                                  /* Whether a space separates currency_symbol from a nonnegative value. */
n_cs_precedes                                                                   /* Whether currency_symbol precedes a negative value. */
n_sep_by_space                                                                  /* Whether a space separates currency_symbol from a negative value. */
p_sign_posn                                                                     /* Position of positive_sign for a nonnegative value. */
n_sign_posn                                                                     /* Position of negative_sign for a negative value. */

//|----------------------------------------------------------------------------
//| Nonlocal Jumps (setjmp.h)
//|----------------------------------------------------------------------------

int           setjmp        (jmp_buf env);                                       /* Saves the calling environment into env for later use by longjmp; returns 0 on this direct call. */
void          longjmp       (jmp_buf env, int val);                              /* Restores the environment saved by setjmp in env, causing the corresponding setjmp call to return val (or 1 if val is 0). */


//|----------------------------------------------------------------------------
//| Types (setjmp.h)
//|----------------------------------------------------------------------------

jmp_buf                                                                         /* Array type suitable for holding the information needed to restore a calling environment. */

//|----------------------------------------------------------------------------
//| String Copying (string.h)
//|----------------------------------------------------------------------------

void         *memcpy        (void *s1, const void *s2, size_t n);                /* Copies n bytes from s2 to s1; behavior is undefined if the objects overlap. */
void         *memmove       (void *s1, const void *s2, size_t n);                /* Copies n bytes from s2 to s1, correctly handling overlapping objects. */
char         *strcpy        (char *s1, const char *s2);                         /* Copies the string s2, including the terminating null character, into s1. */
char         *strncpy       (char *s1, const char *s2, size_t n);               /* Copies at most n characters from s2 into s1, padding with nulls if s2 is shorter than n. */


//|----------------------------------------------------------------------------
//| String Concatenation (string.h)
//|----------------------------------------------------------------------------

char         *strcat        (char *s1, const char *s2);                         /* Appends a copy of s2 to the end of s1. */
char         *strncat       (char *s1, const char *s2, size_t n);               /* Appends at most n characters from s2 to the end of s1, then null-terminates. */


//|----------------------------------------------------------------------------
//| String/Memory Comparison (string.h)
//|----------------------------------------------------------------------------

int           memcmp        (const void *s1, const void *s2, size_t n);          /* Compares the first n bytes of s1 and s2. */
int           strcmp        (const char *s1, const char *s2);                    /* Compares the strings s1 and s2. */
int           strcoll       (const char *s1, const char *s2);                    /* Compares s1 and s2 according to the current locale's collation rules. */
int           strncmp       (const char *s1, const char *s2, size_t n);          /* Compares at most n characters of s1 and s2. */
size_t        strxfrm       (char *s1, const char *s2, size_t n);                /* Transforms s2 into s1 such that strcmp on the results is equivalent to strcoll on the originals. */


//|----------------------------------------------------------------------------
//| String/Memory Searching (string.h)
//|----------------------------------------------------------------------------

void         *memchr        (const void *s, int c, size_t n);                    /* Locates the first occurrence of c in the first n bytes of s. */
char         *strchr        (const char *s, int c);                             /* Locates the first occurrence of c in the string s. */
size_t        strcspn       (const char *s1, const char *s2);                    /* Computes the length of the initial segment of s1 consisting of characters not in s2. */
char         *strpbrk       (const char *s1, const char *s2);                   /* Locates the first character in s1 that matches any character in s2. */
char         *strrchr       (const char *s, int c);                             /* Locates the last occurrence of c in the string s. */
size_t        strspn        (const char *s1, const char *s2);                    /* Computes the length of the initial segment of s1 consisting entirely of characters in s2. */
char         *strstr        (const char *s1, const char *s2);                   /* Locates the first occurrence of the string s2 within s1. */
char         *strtok        (char *s1, const char *s2);                         /* Breaks s1 into a sequence of tokens delimited by characters from s2. */


//|----------------------------------------------------------------------------
//| Miscellaneous (string.h)
//|----------------------------------------------------------------------------

void         *memset        (void *s, int c, size_t n);                         /* Sets the first n bytes of s to the value c. */
char         *strerror      (int errnum);                                       /* Returns a pointer to a string describing the error code errnum. */
size_t        strlen        (const char *s);                                     /* Computes the length of the string s, excluding the terminating null character. */


//|----------------------------------------------------------------------------
//| Constants (string.h)
//|----------------------------------------------------------------------------

NULL                                                                            /* Null pointer constant. */

//| Types (string.h)
//|----------------------------------------------------------------------------

size_t                                                                          /* Unsigned integer type for the result of the sizeof operator. */

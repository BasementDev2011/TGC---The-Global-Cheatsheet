//|----------------------------------------------------------------------------
//| Signal Set Operations (signal.h) [POSIX]
//|----------------------------------------------------------------------------

int           sigemptyset   (sigset_t *set);                                     /* Initializes a signal set to exclude all signals. */
int           sigfillset    (sigset_t *set);                                     /* Initializes a signal set to include all signals. */
int           sigaddset     (sigset_t *set, int signo);                          /* Adds a signal to a signal set. */
int           sigdelset     (sigset_t *set, int signo);                          /* Removes a signal from a signal set. */
int           sigismember   (const sigset_t *set, int signo);                    /* Tests whether a signal is a member of a signal set. */


//|----------------------------------------------------------------------------
//| Signal Actions and Masking (signal.h) [POSIX]
//|----------------------------------------------------------------------------

int           sigaction     (int sig, const struct sigaction *restrict act, struct sigaction *restrict oact); /* Examines or sets the action for a signal. */
int           sigprocmask   (int how, const sigset_t *restrict set, sigset_t *restrict oset); /* Examines or changes the calling process's signal mask. */
int           sigpending    (sigset_t *set);                                     /* Retrieves the set of pending signals blocked from delivery. */
int           sigsuspend    (const sigset_t *sigmask);                           /* Replaces signal mask and suspends the process until a signal is caught. */
int           sigwait       (const sigset_t *restrict set, int *restrict sig);   /* Waits for and removes one of a set of pending signals. */
int           sigwaitinfo   (const sigset_t *restrict set, siginfo_t *restrict info); /* Waits for a signal, returning information about it. */
int           sigtimedwait  (const sigset_t *restrict set, siginfo_t *restrict info, const struct timespec *restrict timeout); /* Waits for a signal with a timeout. */
int           sigqueue      (pid_t pid, int signo, union sigval value);          /* Queues a signal with data to a process. */
int           sigaltstack   (const stack_t *restrict ss, stack_t *restrict oss); /* Sets or retrieves the alternate signal stack. */
int           pthread_sigmask (int how, const sigset_t *restrict set, sigset_t *restrict oset); /* Examines or changes the calling thread's signal mask. */
int           pthread_kill  (pthread_t thread, int sig);                         /* Sends a signal to a thread. */

int           killpg        (pid_t pgrp, int sig);                               /* Sends a signal to a process group. */

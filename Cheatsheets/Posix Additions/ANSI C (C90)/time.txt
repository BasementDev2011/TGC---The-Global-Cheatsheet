//|----------------------------------------------------------------------------
//| Clocks (time.h) [POSIX.1-2001]
//|----------------------------------------------------------------------------

int           clock_getres  (clockid_t clock_id, struct timespec *res);          /* Retrieves the resolution of a clock. */
int           clock_gettime (clockid_t clock_id, struct timespec *tp);           /* Retrieves the current time of a clock. */
int           clock_settime (clockid_t clock_id, const struct timespec *tp);     /* Sets the current time of a clock. */
int           clock_nanosleep (clockid_t clock_id, int flags, const struct timespec *rqtp, struct timespec *rmtp); /* Suspends execution relative to a clock. */
int           nanosleep     (const struct timespec *rqtp, struct timespec *rmtp); /* Suspends execution for a nanosecond interval. */

int           timer_delete  (timer_t timerid);                                   /* Deletes a per-process timer. */
int           timer_settime (timer_t timerid, int flags, const struct itimerspec *restrict value, struct itimerspec *restrict ovalue); /* Arms or disarms a timer. */
int           timer_gettime (timer_t timerid, struct itimerspec *value);         /* Retrieves the remaining time on a timer. */
int           timer_getoverrun (timer_t timerid);                                /* Returns the overrun count for a timer. */
